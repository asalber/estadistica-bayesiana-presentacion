$n \in \mathbb{N},\; p \in (0,1)$ &
$np$ &

\rowcolor{white}
$\text{Geométrica}(p)$ &
$p \in (0,1)$ &
$\dfrac{1}{p}$ &

\rowcolor{rowlight}
$\text{Bin. neg.}(r,p)$ &
$r > 0,\; p \in (0,1)$ &
$\dfrac{r(1-p)}{p}$ &
$\dfrac{r(1-p)}{p^2}$ &
Conteos sobredispersos: recaídas, reingresos, eventos adversos con varianza $>$ media \\

\rowcolor{white}

$\lambda > 0$ &
$\lambda$ &

\rowcolor{rowlight}
$\text{Hipergeo.}(N,K,n)$ &
$N,K,n \in \mathbb{N}$ &
$\dfrac{nK}{N}$ &
$\dfrac{nK(N-K)(N-n)}{N^2(N-1)}$ &
Muestreo sin reemplazamiento: prevalencia en poblaciones pequeñas \\

\rowcolor{headerCont}
\color{white}\textbf{Distribución} &
\color{white}\textbf{Parámetros} &
\color{white}\textbf{Media} &
\color{white}\textbf{Varianza} &
\color{white}\textbf{Usos habituales} \\
\toprule

\rowcolor{white}
$\text{Uniforme}(a,b)$ &
$a < b$ &
$\dfrac{a+b}{2}$ &
$\dfrac{(b-a)^2}{12}$ &
Prior no informativo para parámetros acotados: probabilidades, proporciones \\

\rowcolor{rowlight2}
$\text{Beta}(\alpha,\beta)$ &
$\alpha,\beta > 0$ &
$\dfrac{\alpha}{\alpha+\beta}$ &
$\dfrac{\alpha\beta}{(\alpha+\beta)^2(\alpha+\beta+1)}$ &
Prior conjugado para proporciones: prevalencia, sensibilidad, especificidad \\

\rowcolor{white}
$\text{Dirichlet}(\alpha_i)$ &
$\alpha_i > 0$ &
$\dfrac{\alpha_i}{\alpha_0}$ &
$\dfrac{\alpha_i(\alpha_0-\alpha_i)}{\alpha_0^2(\alpha_0+1)}$ &
Prior conjugado para vectores de probabilidad: serotipos, mezclas de poblaciones \\

\rowcolor{rowlight2}
$\text{Normal}(\mu,\sigma^2)$ &
$\mu\in\mathbb{R},\;\sigma^2>0$ &
$\mu$ &
$\sigma^2$ &
Prior para medias y coeficientes: presión arterial, colesterol, efectos de tratamiento \\

\rowcolor{white}
$t(\nu,\mu,\sigma^2)$ &
$\nu>0,\;\mu\in\mathbb{R},\;\sigma^2>0$ &
$\mu\;(\nu>1)$ &
$\dfrac{\nu\sigma^2}{\nu-2}\;(\nu>2)$ &
Prior robusto con colas pesadas; marginal de $\mu$ en modelo Normal-Inv-Gamma \\

\rowcolor{rowlight2}
$\text{Cauchy}(\mu,\sigma)$ &
$\mu\in\mathbb{R},\;\sigma>0$ &
no existe &
no existe &
Prior de colas muy pesadas para coeficientes en regresión logística \\

\rowcolor{white}
$\text{Exponencial}(\lambda)$ &
$\lambda > 0$ &
$\dfrac{1}{\lambda}$ &
$\dfrac{1}{\lambda^2}$ &
Prior débilmente informativo para $\sigma$ y $\tau$; tiempo entre eventos con riesgo constante \\

\rowcolor{rowlight2}
$\text{Gamma}(\alpha,\beta)$ &
$\alpha,\beta > 0$ &
$\dfrac{\alpha}{\beta}$ &
$\dfrac{\alpha}{\beta^2}$ &
Prior conjugado para tasas Poisson: incidencia, mortalidad, $R(t)$ \\

\rowcolor{white}
$\text{Gamma}^{-1}(\alpha,\beta)$ &
$\alpha,\beta > 0$ &
$\dfrac{\beta}{\alpha-1}\;(\alpha>1)$ &
$\dfrac{\beta^2}{(\alpha-1)^2(\alpha-2)}\;(\alpha>2)$ &
Prior conjugado para varianzas $\sigma^2$; varianza de efectos aleatorios \\

\rowcolor{rowlight2}
$\text{Lognormal}(\mu,\sigma^2)$ &
$\mu\in\mathbb{R},\;\sigma^2>0$ &
$e^{\mu+\sigma^2/2}$ &
$(e^{\sigma^2}-1)e^{2\mu+\sigma^2}$ &
Parámetros positivos asimétricos: dosis, concentraciones, tiempos de supervivencia \\

\rowcolor{white}
$\text{Weibull}(k,\lambda)$ &
$k,\lambda > 0$ &
$\lambda\,\Gamma\!\left(1+\tfrac{1}{k}\right)$ &
$\lambda^2\!\left[\Gamma\!\left(1+\tfrac{2}{k}\right) - \Gamma\!\left(1+\tfrac{1}{k}\right)^{\!2}\right]$ &
Modelos de supervivencia: tiempo hasta muerte, recaída o fracaso de tratamiento \\

\rowcolor{rowlight2}
$\text{NIGamma}(\mu_0,\lambda,\alpha,\beta)$ &
$\mu_0\in\mathbb{R},\;\lambda,\alpha,\beta>0$ &
$\left(\mu_0,\;\dfrac{\beta}{\alpha-1}\right)$ &
--- &
Prior conjugado conjunto para $(\mu, \sigma^2)$; base de la regresión bayesiana \\

\bottomrule
\end{tabular}
